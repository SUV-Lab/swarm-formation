cmake_minimum_required(VERSION 3.16)
project(mmp_exact_mission_consumer LANGUAGES CXX)
find_package(path_manager REQUIRED)
get_target_property(mmp_imported_features path_manager::path_manager_lib INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST mmp_imported_features)
  message(FATAL_ERROR "Installed path_manager target must export cxx_std_17; got: ${mmp_imported_features}")
endif()
message(STATUS "Verified path_manager exported compile features: ${mmp_imported_features}")
add_executable(exact_mission_consumer main.cpp)
target_link_libraries(exact_mission_consumer PRIVATE path_manager::path_manager_lib)
set_target_properties(exact_mission_consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)

add_library(facade_header_probe OBJECT facade_header_probe.cpp)
target_link_libraries(facade_header_probe PRIVATE path_manager::path_manager_lib)
target_compile_options(facade_header_probe PRIVATE -H)
set_target_properties(facade_header_probe PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
