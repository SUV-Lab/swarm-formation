cmake_minimum_required(VERSION 3.16)
project(mmp_optimizer_only_consumer LANGUAGES CXX)
find_package(path_optimizer REQUIRED)
get_target_property(optimizer_features path_optimizer::path_optimizer_lib INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST optimizer_features)
  message(FATAL_ERROR "Optimizer target must export cxx_std_17: ${optimizer_features}")
endif()
message(STATUS "Verified optimizer exported compile features: ${optimizer_features}")
add_executable(optimizer_only_consumer main.cpp)
target_link_libraries(optimizer_only_consumer PRIVATE path_optimizer::path_optimizer_lib)
